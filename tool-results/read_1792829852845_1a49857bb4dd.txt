    87→    87→function FounderPanel({
    88→    88→  ctx,
    89→    89→  unlocked,
    90→    90→  onClearAllChats,
    91→    91→}: {
    92→    92→  ctx: ContextResponse | null;
    93→    93→  unlocked: boolean;
    94→    94→  onClearAllChats: () => void;
    95→    95→}) {
    96→    96→  const [pin, setPin] = useState('');
    97→    97→  const [pinError, setPinError] = useState<string | null>(null);
    98→    98→  const [config, setConfig] = useState<FounderConfig | null>(null);
    99→    99→  const [newPin, setNewPin] = useState('');
   100→   100→  const [pinMsg, setPinMsg] = useState<string | null>(null);
   101→   101→  const [keyInput, setKeyInput] = useState('');
   102→   102→  const [baseUrl, setBaseUrl] = useState('');
   103→   103→  const [modelId, setModelId] = useState('');
   104→   104→  const [builtinModel, setBuiltinModel] = useState('glm-4-plus');
   105→   105→  const [provider, setProvider] = useState<'builtin' | 'custom'>('builtin');
   106→   106→  const [testing, setTesting] = useState(false);
   107→   107→  const [testResult, setTestResult] = useState<{ ok: boolean; msg: string } | null>(null);
   108→   108→  const [saving, setSaving] = useState(false);
   109→   109→  const [saveMsg, setSaveMsg] = useState<string | null>(null);
   110→   110→
   111→   111→  const unlock = async () => {
   112→   112→    setPinError(null);
   113→   113→    const res = await fetch('/api/founder/unlock', {
   114→   114→      method: 'POST',
   115→   115→      headers: { 'Content-Type': 'application/json' },
   116→   116→      body: JSON.stringify({ pin }),
   117→   117→    });
   118→   118→    if (!res.ok) {
   119→   119→      setPinError('Incorrect PIN. Try again.');
   120→   120→      setPin('');
   121→   121→      return;
   122→   122→    }
   123→   123→    const j = await res.json();
   124→   124→    applyConfig(j.config);
   125→   125→    sessionStorage.setItem('saathy_founder_pin', pin);
   126→   126→  };
   127→   127→
   128→   128→  const applyConfig = (c: FounderConfig) => {
   129→   129→    setConfig(c);
   130→   130→    setProvider(c.provider === 'custom' ? 'custom' : 'builtin');
   131→   131→    setBaseUrl(c.customBaseUrl ?? '');
   132→   132→    setModelId(c.customModelId ?? '');
   133→   133→    setBuiltinModel(c.builtinModelId || 'glm-4-plus');
   134→   134→  };
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    const savedPin = sessionStorage.getItem('saathy_founder_pin');
   138→   138→    if (savedPin && !config) {
   139→   139→      setPin(savedPin);
   140→   140→      fetch('/api/founder/unlock', {
   141→   141→        method: 'POST',
   142→   142→        headers: { 'Content-Type': 'application/json' },
   143→   143→        body: JSON.stringify({ pin: savedPin }),
   144→   144→      })
   145→   145→        .then((r) => (r.ok ? r.json() : null))
   146→   146→        .then((j) => {
   147→   147→          if (j?.config) applyConfig(j.config);
   148→   148→          else sessionStorage.removeItem('saathy_founder_pin');
   149→   149→        })
   150→   150→        .catch(() => sessionStorage.removeItem('saathy_founder_pin'));
   151→   151→    }
   152→   152→    // eslint-disable-next-line react-hooks/exhaustive-deps
   153→   153→  }, []);
   154→   154→
   155→   155→  const saveConfig = async () => {
   156→   156→    setSaving(true);
   157→   157→    setSaveMsg(null);
   158→   158→    try {
   159→   159→      const res = await fetch('/api/founder/config', {
   160→   160→        method: 'POST',
   161→   161→        headers: { 'Content-Type': 'application/json' },
   162→   162→        body: JSON.stringify({
   163→   163→          pin,
   164→   164→          provider,
   165→   165→          builtinModelId: builtinModel,
   166→   166→          customBaseUrl: baseUrl,
   167→   167→          customModelId: modelId,
   168→   168→          ...(keyInput.trim() ? { customApiKey: keyInput.trim() } : {}),
   169→   169→        }),
   170→   170→      });
   171→   171→      const j = await res.json();
   172→   172→      if (!res.ok || !j.ok) {
   173→   173→        setSaveMsg(j.error ?? 'Could not save configuration');
   174→   174→        return;
   175→   175→      }
   176→   176→      applyConfig(j.config);
   177→   177→      setKeyInput('');
   178→   178→      setSaveMsg('Saved. Verifying connection…');
   179→   179→
   180→   180→      // Real verification: a genuine request right after saving
   181→   181→      const testRes = await fetch('/api/founder/test', {
   182→   182→        method: 'POST',
   183→   183→        headers: { 'Content-Type': 'application/json' },
   184→   184→        body: JSON.stringify({ pin, target: provider === 'custom' ? 'custom' : 'builtin' }),
   185→   185→      });
   186→   186→      const tj = await testRes.json();
   187→   187→      if (tj.ok) {
   188→   188→        setTestResult({ ok: true, msg: `Connected in ${tj.latencyMs}ms · ${tj.model ?? tj.reply ?? ''}` });
   189→   189→        setSaveMsg('Saved & verified. Saathy is running on this configuration.');
   190→   190→        // re-fetch config to show verifiedAt
   191→   191→        const cfgRes = await fetch('/api/founder/config', {
   192→   192→          method: 'POST',
   193→   193→          headers: { 'Content-Type': 'application/json' },
   194→   194→          body: JSON.stringify({ pin }),
   195→   195→        });
   196→   196→        if (cfgRes.ok) {
   197→   197→          const cj = await cfgRes.json();
   198→   198→          if (cj.config) applyConfig(cj.config);
   199→   199→        }
   200→   200→      } else {
   201→   201→        setTestResult({ ok: false, msg: tj.error ?? 'Connection failed' });
   202→   202→        setSaveMsg('Saved, but the connection test failed — see the result below.');
   203→   203→      }
   204→   204→    } finally {
   205→   205→      setSaving(false);
   206→   206→    }
   207→   207→  };
   208→   208→
   209→   209→  const testOnly = async () => {

   211→   211→    setTestResult(null);
   212→   212→    try {
   213→   213→      const res = await fetch('/api/founder/test', {
   214→   214→        method: 'POST',
   215→   215→        headers: { 'Content-Type': 'application/json' },
   216→   216→        body: JSON.stringify(
   217→   217→          provider === 'custom'
   218→   218→            ? { pin, target: 'custom', customBaseUrl: baseUrl, customApiKey: keyInput.trim() || undefined, customModelId: modelId }
   219→   219→            : { pin, target: 'builtin' }
   220→   220→        ),
   221→   221→      });
   222→   222→      const j = await res.json();
   223→   223→      setTestResult(j.ok ? { ok: true, msg: `Connected in ${j.latencyMs}ms · ${j.model ?? j.reply ?? ''}` } : { ok: false, msg: j.error ?? 'Connection failed' });
   224→   224→    } finally {
   225→   225→      setTesting(false);
   226→   226→    }
   227→   227→  };
   228→   228→
   229→   229→  const changePin = async () => {

   232→   232→      setPinMsg('PIN must be exactly 4 digits.');

   234→   234→    }
   235→   235→    const res = await fetch('/api/founder/config', {
   236→   236→      method: 'POST',
   237→   237→      headers: { 'Content-Type': 'application/json' },
   238→   238→      body: JSON.stringify({ pin, newPin }),
   239→   239→    });
   240→   240→    const j = await res.json();
   241→   241→    if (!res.ok || !j.ok) {
   242→   242→      setPinMsg(j.error ?? 'Could not change PIN');
   243→   243→      return;
   244→   244→    }
   245→   245→    setPinMsg('PIN changed. New PIN is active now.');
   246→   246→    setNewPin('');
   247→   247→    sessionStorage.setItem('saathy_founder_pin', newPin);
   248→   248→    setPin(newPin);
   249→   249→  };
   250→   250→
   251→   251→  const deleteKey = async () => {
   252→   252→    const res = await fetch(`/api/founder/config?pin=${encodeURIComponent(pin)}`, { method: 'DELETE' });
   253→   253→    const j = await res.json();
   254→   254→    if (j.ok) {
   255→   255→      applyConfig(j.config);
   256→   256→      setKeyInput('');
   257→   257→      setSaveMsg('Stored API key deleted.');
   258→   258→    }
   259→   259→  };
   260→   260→
   261→   261→  if (!unlocked && !config) {
   262→   262→    return (
   263→   263→      <div className="space-y-3">
   264→   264→        <div className="rounded-xl border border-[#ffe0c7] bg-[#fff3ea] p-4 text-center">
   265→   265→          <Mascot pose="thinking" size={64} className="mx-auto" />
   266→   266→          <p className="mt-1 flex items-center justify-center gap-1.5 text-[13.5px] font-extrabold text-[#7a3b1d]">
   267→   267→            <Lock className="h-3.5 w-3.5" /> Founder access
   268→   268→          </p>
   269→   269→          <p className="mx-auto mt-1 max-w-[36ch] text-[11.5px] font-semibold leading-relaxed text-[#a05c2e]">

   275→   275→            Founder PIN

   288→   288→          {pinError ? <p className="text-center text-[12px] font-bold text-[#e5484d]">{pinError}</p> : null}

   290→   290→            Unlock
   291→   291→          </Button>
   292→   292→          <p className="text-center text-[10.5px] font-semibold leading-relaxed text-[#b3bdcf]">
   293→   293→            Prototype note: this PIN is a convenience gate, not production security. In the VidhyaSaathy app this area sits behind server-side admin roles.

   295→   295→        </div>
   296→   296→      </div>

   305→   305→      <div className="rounded-xl border border-[#ece7e0] bg-white p-3.5">
   306→   306→        <p className="mb-2 flex items-center gap-1.5 text-[12px] font-extrabold uppercase tracking-wider text-[#8a94a8]">

   309→   309→        <SegGroup
   310→   310→          value={provider}
   311→   311→          onChange={(v) => setProvider(v)}
   312→   312→          options={[
   313→   313→            { value: 'builtin', label: 'Built-in Z.ai GLM' },
   314→   314→            { value: 'custom', label: 'Custom (OpenAI-compatible)' },
   315→   315→          ]}
   316→   316→        />
   317→   317→
   318→   318→        {provider === 'builtin' ? (
   319→   319→          <div className="mt-3 space-y-2">
   320→   320→            <Label className="text-[11px] font-extrabold uppercase tracking-wider text-[#8a94a8]">Model ID</Label>

   331→   331→            </div>
   332→   332→            <div>
   333→   333→              <Label className="text-[11px] font-extrabold uppercase tracking-wider text-[#8a94a8]">API key</Label>
   334→   334→              {config.customApiKeySet ? (

   336→   336→                  <Input readOnly value={config.customApiKeyMasked || 'Stored'} className="h-9 text-[13px] font-bold" />
   337→   337→                  <Button type="button" variant="outline" size="sm" onClick={deleteKey} className="h-9 shrink-0 gap-1 border-[#f3c1c2] text-[12px] font-bold text-[#9c2c30]">

   342→   342→              <Input

   344→   344→                value={keyInput}
