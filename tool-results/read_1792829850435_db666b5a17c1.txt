    98→  const [config, setConfig] = useState<FounderConfig | null>(null);
    99→  const [newPin, setNewPin] = useState('');
   100→  const [pinMsg, setPinMsg] = useState<string | null>(null);
   101→  const [keyInput, setKeyInput] = useState('');
   102→  const [baseUrl, setBaseUrl] = useState('');
   103→  const [modelId, setModelId] = useState('');
   104→  const [builtinModel, setBuiltinModel] = useState('glm-4-plus');
   105→  const [provider, setProvider] = useState<'builtin' | 'custom'>('builtin');
   106→  const [testing, setTesting] = useState(false);
   107→  const [testResult, setTestResult] = useState<{ ok: boolean; msg: string } | null>(null);
   108→  const [saving, setSaving] = useState(false);
   109→  const [saveMsg, setSaveMsg] = useState<string | null>(null);
   110→
   111→  const unlock = async () => {
   112→    setPinError(null);
   113→    const res = await fetch('/api/founder/unlock', {
   114→      method: 'POST',
   115→      headers: { 'Content-Type': 'application/json' },
   116→      body: JSON.stringify({ pin }),
   117→    });

   121→      return;

   127→
   128→  const applyConfig = (c: FounderConfig) => {

   141→        method: 'POST',
   142→        headers: { 'Content-Type': 'application/json' },
   143→        body: JSON.stringify({ pin: savedPin }),

   155→  const saveConfig = async () => {

   157→    setSaveMsg(null);
   158→    try {
   159→      const res = await fetch('/api/founder/config', {
   160→        method: 'POST',
   161→        headers: { 'Content-Type': 'application/json' },
   162→        body: JSON.stringify({
   163→          pin,
   164→          provider,
   165→          builtinModelId: builtinModel,
   166→          customBaseUrl: baseUrl,

   175→      }

   177→      setKeyInput('');
   178→      setSaveMsg('Saved. Verifying connection…');

   180→      // Real verification: a genuine request right after saving
   181→      const testRes = await fetch('/api/founder/test', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({ pin, target: provider === 'custom' ? 'custom' : 'builtin' }),
   185→      });

   188→        setTestResult({ ok: true, msg: `Connected in ${tj.latencyMs}ms · ${tj.model ?? tj.reply ?? ''}` });
   189→        setSaveMsg('Saved & verified. Saathy is running on this configuration.');

   193→          headers: { 'Content-Type': 'application/json' },

   290→            Unlock

   316→        />
   317→

   437→export function SettingsSheet({
   438→  open,
   439→  onOpenChange,
   440→  initialTab,
   441→  ctx,
   442→  founderUnlocked,
   443→  onPatchProfile,
   444→  onConceptFocus,
   445→  onClearAllChats,
   446→}: SettingsSheetProps) {
   447→  const [tab, setTab] = useState<'chat' | 'student' | 'founder'>(initialTab);
   448→  const [subjectId, setSubjectId] = useState<string>('');
   449→  const [chapterId, setChapterId] = useState<string>('');
   450→
   451→  useEffect(() => {
   452→    if (open) setTab(initialTab);
   453→  }, [open, initialTab]);
   454→
   455→  useEffect(() => {
   456→    if (!ctx) return;
   457→    setSubjectId(ctx.profile.focusSubjectId ?? ctx.subjects[0]?.id ?? '');
   458→    setChapterId(ctx.profile.focusChapterId ?? ctx.subjects[0]?.chapters[0]?.id ?? '');
   459→  }, [ctx, open]);
   460→
   461→  const profile = ctx?.profile;
   462→  const subjects: SubjectCtx[] = ctx?.subjects ?? [];
   463→  const subject = subjects.find((s) => s.id === subjectId);
   464→  const chapter = subject?.chapters.find((c) => c.id === chapterId);
   465→
